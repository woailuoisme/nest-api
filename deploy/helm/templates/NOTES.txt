1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "nestforge.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  Watch the status with: kubectl get svc --namespace {{ .Release.Namespace }} -w {{ include "nestforge.fullname" . }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "nestforge.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "nestforge.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:4300 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 4300:{{ .Values.service.port }}
{{- end }}

2. Documentation & monitoring endpoints:
{{- if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}
  Swagger API Docs: http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}/docs
  Health Check:      http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}/health
  Metrics:           http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}/metrics
{{- end }}
{{- else }}
  Swagger API Docs: /docs
  Health Check:      /health
  Metrics:           /metrics
{{- end }}

3. Run database migrations (after ensuring PostgreSQL is accessible):
   kubectl exec --namespace {{ .Release.Namespace }} -it deploy/{{ include "nestforge.fullname" . }} -- bun x drizzle-kit migrate

4. Seed database (optional):
   kubectl exec --namespace {{ .Release.Namespace }} -it deploy/{{ include "nestforge.fullname" . }} -- bun x tsx src/database/seeds/index.ts
